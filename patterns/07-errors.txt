swallowed-exception	py	except[^:\n]*:[ \t]*(\n[ \t]+)?(pass|\.\.\.)[ \t]*(\n|$)		never swallow an exception — log it and fail closed, or let it propagate	high
swallowed-exception	js,ts,java,kt,cs,php	catch\s*\([^)]*\)\s*\{\s*\}		empty catch block hides the failure — log and re-throw, or handle explicitly	high
swallowed-exception	rb	rescue[^\n]*\n\s*end		empty rescue hides the failure — log and re-raise, or handle explicitly	high
