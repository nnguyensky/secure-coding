sbd-legacy-tls	py,js,ts,go,java,c,cpp	(PROTOCOL_TLSv1_1|PROTOCOL_TLSv1_0|PROTOCOL_SSLv3|TLSv1\.0|TLSv1\.1|SSLv3)		deprecated legacy TLS/SSL protocol version; mandate TLSv1.2 or TLSv1.3	high
sbd-dynamic-eval-reflection	py,js,ts	(eval|exec)\s*\([^)]*(req\.|request\.|params\.|body\.|query\.)		avoid dynamic code evaluation on untrusted request inputs	critical
sbd-unauthenticated-route	py,js,ts	(router|app)\.(get|post|put|delete)\s*\(\s*["']/admin/	!auth|authenticate|require_auth|jwt	administrative route missing authentication or authorization guard	high
solidity-tx-origin	sol	\btx\.origin\s*(==|!=)|require\s*\(\s*tx\.origin	authorize with msg.sender and AccessControl		critical
solidity-unchecked-call	sol	\.call\s*\{\s*value\s*:[^}]*\}\s*\(\s*["']{2}\s*\)\s*;|\.send\s*\(\s*[^)]*\)\s*;	!\(bool|require|if\s*\(	check the returned bool, update state first, and guard with nonReentrant	high
solidity-delegatecall	sol	\.delegatecall\s*\(	never delegatecall into an address a caller can choose		critical
solidity-weak-randomness	sol	keccak256\s*\([^;]*(block\.(timestamp|number|prevrandao|difficulty)|blockhash)	use a VRF or commit-reveal		high
solidity-old-compiler	sol	pragma\s+solidity\s*[\^>=<~]*\s*0\.[4-7]\.	use 0.8.x (checked arithmetic) and pin the exact version		high
solidity-selfdestruct	sol	\bselfdestruct\s*\(	remove selfdestruct; it is deprecated and dangerous in upgradeable systems		high
